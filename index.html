# 学生成绩管理系统（简化版） students = [] def add_student(name, score): students.append({"name": name, "score": score}) def show_stats(): if not students: print("暂无数据") return scores = [s["score"] for s in students] print(f"平均分: {sum(scores)/len(scores):.2f}") print(f"最高分: {max(scores)}, 最低分: {min(scores)}") add_student("陈依琳", 95) add_student("同学A", 88) show_stats()
